    def create_training_data(ratings_path, movies_path, num_users=500):
        """Convert user ratings to conversation-like text for recommender training."""
        ratings_df = pd.read_csv(ratings_path)
        movies_df = pd.read_csv(movies_path)
        
        conversations = []
        liked_movies = []
        sampled_users = ratings_df['userId'].unique()[:num_users]
        
        for user_id in sampled_users:
            user_ratings = ratings_df[ratings_df['userId'] == user_id]
            high_ratings = user_ratings[user_ratings['rating'] >= 4.0]
            
            if len(high_ratings) < 3:
                continue
            
            sampled = high_ratings.sample(min(5, len(high_ratings)))
            movie_ids = sampled['movieId'].tolist()
            
            titles = []
            for mid in movie_ids[:3]:
                movie_row = movies_df[movies_df['movieId'] == mid]
                if len(movie_row) > 0:
                    title = movie_row.iloc[0]['title'].split('(')[0].strip()
                    titles.append(title)
            
            if len(titles) >= 2:
                conv = f"I enjoyed {titles[0]} and {titles[1]}"
                if len(titles) > 2:
                    conv += f", also {titles[2]}"
                conversations.append(conv)
                liked_movies.append(movie_ids)
        
        return conversations, liked_movies

    convs, likes = create_training_data(
        DATA_DIR / 'ratings.csv',
        DATA_DIR / 'movies.csv',
        num_users=500
    )
    
    print(f"Training recommender on {len(convs)} examples...")

    }, rec_checkpoint)

            # Supervised pretraining: predict concept embeddings from conversation state
            optimizer = torch.optim.Adam(actor_critic.actor.parameters(), lr=0.001)
            
            for epoch in range(PRETRAIN_EPOCHS):
                epoch_loss = 0
                
                for batch in train_data:
                    state = torch.FloatTensor(batch['state'])
                    target_embedding = torch.FloatTensor(batch['target_embedding'])
                    
                    predicted_embedding = actor_critic.actor(state)
                    loss = torch.nn.functional.mse_loss(predicted_embedding, target_embedding)
                    
                    optimizer.zero_grad()
                    loss.backward()
                    optimizer.step()
                    
                    epoch_loss += loss.item()
                
                avg_loss = epoch_loss / len(train_data)
                print(f"Pretraining epoch {epoch+1}/{PRETRAIN_EPOCHS}: loss={avg_loss:.4f}")
                
                # Save checkpoint every 5 epochs
                if (epoch + 1) % 5 == 0:
                    torch.save({
                        'actor': actor_critic.actor.state_dict(),
                        'epoch': epoch + 1,
                        'loss': avg_loss
                    }, pretrain_checkpoint)
            
            print(f"Pretraining complete. Checkpoint saved: {pretrain_checkpoint}")
        else:
            print("No Reddit training data found. Skipping supervised pretraining.")
    else:
        print(f"Reddit data directory not found: {reddit_data_dir}")

def run_episode(actor_critic, embedding_space, user_profile, evaluator, max_turns=10):
    """Run single episode and return rewards + final NDCG."""
    from casper.models.question_generator import QuestionGenerator
    from casper.models.preference_extractor import PreferenceExtractor
    
    question_gen = QuestionGenerator()
    pref_extractor = PreferenceExtractor()
    
    training_profile, test_set = evaluator.prepare_user_profile(user_profile)
    
    conversation_history = []
    discovered_preferences = {"liked": [], "neutral": [], "disliked": []}
    rewards = []
    prev_ndcg = 0.0
    
    for turn in range(max_turns):
        # Encode state
        pref_text = pref_extractor.preferences_to_text(discovered_preferences)
        state = embedding_space.encode_text(pref_text) if pref_text else torch.zeros(384)
        
        # Actor predicts action
        action = actor_critic.select_action(state, noise=0.2)
        
        # Map to entities
        entities = embedding_space.find_nearest_entities(action, k=5)
        
        # Generate question
        question = question_gen.generate_question(
            entities=entities,
            conversation_history=conversation_history,
            discovered_preferences=discovered_preferences
        )
        
        # Get user response
        response = user_sim.simulate_response(
            question=question,
            user_profile=training_profile,
            conversation_history=conversation_history
        )
        
        conversation_history.append(f"Agent: {question}")
        conversation_history.append(f"User: {response}")
        
        # Extract preferences
        discovered_preferences = pref_extractor.extract_from_conversation(
            conversation_history,
            existing_preferences=discovered_preferences
        )
        
        # Calculate reward (NDCG improvement)
        state_emb = embedding_space.encode_text(pref_extractor.preferences_to_text(discovered_preferences))
        recs = recommender.recommend(state_emb, k=10)
        current_ndcg = metrics.ndcg_at_k(recs, test_set['held_out_movies'], k=10)
        reward = current_ndcg - prev_ndcg
        prev_ndcg = current_ndcg
        
        rewards.append(reward)
    
    return rewards, prev_ndcg
